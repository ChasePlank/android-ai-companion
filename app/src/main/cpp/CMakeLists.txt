cmake_minimum_required(VERSION 3.22.1)
project(alongside_offline LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/offline-llama" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(alongside_offline SHARED offline_ai.cpp)
target_compile_features(alongside_offline PRIVATE cxx_std_17)
target_compile_options(alongside_offline PRIVATE -fexceptions -frtti)
target_link_options(alongside_offline PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(alongside_offline PRIVATE llama android log)
